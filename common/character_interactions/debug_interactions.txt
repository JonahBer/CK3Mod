debug_ping_self = {
	category = interaction_debug_main
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	common_interaction = yes
	icon = debug_secret
	is_shown = { debug_only = yes
	}
	send_options_exclusive = yes
	send_option = {
		flag = ping_option_1
		localization = debug_ping_self_1
	}
	send_option = {
		flag = ping_option_2
		localization = debug_ping_self_2
	}
	on_accept = {
		send_interface_toast = {
			left_icon = scope:actor
			right_icon = scope:recipient
		}
		scope:actor = { add_gold = 50
		}
		scope:actor = { add_prestige = 100
		}
	}
	auto_accept = yes
}
debug_set_slave_decorative_flag_on_character = {
	type = character_interaction
	category = interaction_debug_main
	is_shown = {
		debug_only = yes
		NOT = { scope:recipient = { has_character_flag = is_slave_decorative
			}
		}
	}
	is_valid_showing_failures_only = {
		NOT = { scope:recipient = { has_character_flag = is_slave_decorative
			}
		}
	}
	effect = {
		scope:recipient = { add_character_flag = is_slave_decorative
		}
	}
	ai_potential = { always = no
	}
}
