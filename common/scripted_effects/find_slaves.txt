# common / scripted_effects / find_slaves.txt
# TODO: function for finding nth slave of a flag for having portrait in window_character_slaves.gui
# The idea: the slaves are of 4 different tiers, each with their own character flag. The tiers are: decorative, concubine, secondary wife, great wife.
# This function would find the nth slave of a given tier (character flag) for displaying their portrait in the slave management window.
# The slaves are forced to stay within the owner's court so we need a way to loop through the owners court memebers and find the nth slave of a given tier.
# common/scripted_effects/find_slaves.txt
# Finds the Nth courtier with a given slave tier flag and stores their character ID in a variable on the owner.
# common/scripted_effects/find_slaves.txt
# Finds the Nth courtier with a given slave tier flag and stores their character ID in a variable on the owner.
#
# Variables written (examples):
#   slave_decorative_0 .. slave_decorative_6
#   slave_concubine_0 .. slave_concubine_6
#   slave_secondary_wife_0 .. slave_secondary_wife_2
#   slave_great_wife_0
#
# NOTE: The engine stores `value = this` as a character reference in script, but the GUI data system
# does *not* support looking up a Character by ID directly. Use these variables in script, not in GUI.
find_nth_courtier_with_flag_to_var = {
	# PARAMS:
	#   owner = <character scope>   (usually GetPlayer / scope:actor)
	#   flag  = <character_flag>    (use: is_slave_decorative)
	#   n     = <int>              (1-based)
	#   out_var = <string>         (variable name on owner)
	$owner$ = { save_temporary_scope_as = _owner
	}
	# clear output each run
	scope:_owner = { remove_variable = $out_var$
	}
	# counters on owner
	scope:_owner = {
		set_variable = { name = _match_count value = 0
		}
		set_variable = { name = _found value = 0
		}
	}
	scope:_owner = {
		every_courtier = {
			limit = {
				# stop after found
				scope:_owner = { var:_found < 1
				}
				has_character_flag = $flag$
			}
			scope:_owner = { change_variable = { name = _match_count add = 1
				}
			}
			if = {
				limit = {
					scope:_owner = {
						AND = {
							var:_match_count >= $n$
							var:_match_count <= $n$
						}
					}
				}
				scope:_owner = {
					set_variable = { name = $out_var$ value = this
					}
					set_variable = { name = _found value = 1
					}
				}
			}
		}
	}
	# cleanup
	scope:_owner = { remove_variable = _match_count
	}
	scope:_owner = { remove_variable = _found
	}
}
refresh_slave_slots_decorative = {
	$owner$ = { save_temporary_scope_as = _owner
	}
	find_nth_courtier_with_flag_to_var = { owner = scope:_owner flag = is_slave_decorative n = 1 out_var = slave_decorative_0
	}
	find_nth_courtier_with_flag_to_var = { owner = scope:_owner flag = is_slave_decorative n = 2 out_var = slave_decorative_1
	}
	find_nth_courtier_with_flag_to_var = { owner = scope:_owner flag = is_slave_decorative n = 3 out_var = slave_decorative_2
	}
	find_nth_courtier_with_flag_to_var = { owner = scope:_owner flag = is_slave_decorative n = 4 out_var = slave_decorative_3
	}
	find_nth_courtier_with_flag_to_var = { owner = scope:_owner flag = is_slave_decorative n = 5 out_var = slave_decorative_4
	}
	find_nth_courtier_with_flag_to_var = { owner = scope:_owner flag = is_slave_decorative n = 6 out_var = slave_decorative_5
	}
	find_nth_courtier_with_flag_to_var = { owner = scope:_owner flag = is_slave_decorative n = 7 out_var = slave_decorative_6
	}
}
